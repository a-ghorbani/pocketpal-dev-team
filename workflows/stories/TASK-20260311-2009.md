### Background
PocketPal integrates with PalsHub, a marketplace for AI personas ("pals"). Premium pals require purchase through palshub.ai. Currently, for unowned premium pals, the PalDetailSheet shows only passive informational text ("Visit palshub.ai to explore more pals"). There is no actionable buy button because in-app purchases are not yet implemented. 

Post-May 2025, Apple allows apps to link to external purchase URLs without requiring StoreKit entitlements for US App Store users. This task adds a "Buy on Palshub" deep link button that opens the palshub.ai purchase page in the browser, gated to US-only users. Non-US users continue to see the existing passive text (no behavior change).

### Current State
- `src/components/PalsHub/PalDetailSheet/PalDetailSheet.tsx` (lines 322-327): For unowned premium pals (`palLabel.type === 'premium' && !displayPal.is_owned`), shows an `infoTextContainer` with passive text from `getPremiumInfoText()`.
- `src/utils/palshub-display.ts`: `getPalActionText()` returns `null` for unowned premium pals. `getPremiumInfoText()` returns a non-actionable string.
- No native module exists for storefront/region detection.
- No palshub.ai purchase URL construction logic exists.
- All existing Android native modules use TurboModule pattern: `NativeSpec` base class + `TurboReactPackage`.
- All existing TurboModule specs live in `src/specs/` and use `TurboModuleRegistry`.

### Target State
- New TurboModule spec (`src/specs/NativeStorefront.ts`) defining the `getCountryCode()` interface.
- New native iOS module (`StorefrontModule`) using `SKStorefront` to detect user's App Store country code.
- New native Android module (`StorefrontModule`) extending `NativeStorefrontSpec`, using device locale as a proxy for region.
- Region check runs once at app startup via `PalStore.initialize()`, stored as MobX observable.
- `PalDetailSheet` reads `isUSRegion` synchronously from `palStore` -- no async state, no flash.
- For US users viewing unowned premium pals: "Buy on Palshub" button opens `${PALSHUB_API_BASE_URL}/pals/{palId}` (URL from env config).
- Non-US users continue to see the existing passive informational text.
- Graceful fallback: if native module is unavailable, treat as non-US.

```swift
// StorefrontModule.swift
import Foundation
import React
import StoreKit

@objc(StorefrontModule)
class StorefrontModule: NSObject, RCTBridgeModule {

    @objc
    static func moduleName() -> String! {
        return "StorefrontModule"
    }

    @objc
    static func requiresMainQueueSetup() -> Bool {
        return false
    }

    @objc
    func getCountryCode(_ resolve: @escaping RCTPromiseResolveBlock, rejecter reject: @escaping RCTPromiseRejectBlock) {
        if #available(iOS 16.0, *) {
            // StoreKit 2: returns ISO 3166-1 alpha-2 (e.g., 'US', 'GB')
            Task {
                let storefront = await Storefront.current
                resolve(storefront?.countryCode)
            }
        } else if #available(iOS 13.0, *) {
            // StoreKit 1 fallback: returns ISO 3166-1 alpha-3 (e.g., 'USA', 'GBR')
            let countryCode = SKPaymentQueue.default().storefront?.countryCode
            resolve(countryCode)
        } else {
            resolve(nil)
        }
    }
}

package com.pocketpal

import com.facebook.react.bridge.*
import com.facebook.react.module.annotations.ReactModule
import com.pocketpal.specs.NativeStorefrontSpec
import java.util.Locale

@ReactModule(name = NativeStorefrontSpec.NAME)
class StorefrontModule(reactContext: ReactApplicationContext) :
    NativeStorefrontSpec(reactContext) {

    override fun getName(): String = NativeStorefrontSpec.NAME

    override fun getCountryCode(promise: Promise) {
        try {
            val countryCode = Locale.getDefault().country
            promise.resolve(countryCode)
        } catch (e: Exception) {
            promise.reject("ERROR", e.message)
        }
    }
}
```

// src/utils/region.ts
import NativeStorefront from '../specs/NativeStorefront';

let cachedCountryCode: string | null | undefined; // undefined = not yet fetched

export async function getStorefrontCountryCode(): Promise<string | null> {
  if (cachedCountryCode !== undefined) {
    return cachedCountryCode;
  }

  try {
    if (!NativeStorefront) {
      cachedCountryCode = null;
      return null;
    }
    const code = await NativeStorefront.getCountryCode();
    cachedCountryCode = code ?? null;
    return cachedCountryCode;
  } catch {
    cachedCountryCode = null;
    return null;
  }
}

/**
 * Check if the user is in the US storefront/region.
 * iOS 16+: StoreKit 2 returns alpha-2 ('US').
 * iOS 15: SK1 returns alpha-3 ('USA').
 * Android: Locale returns alpha-2 ('US').
 */
export async function isUSStorefront(): Promise<boolean> {
  const code = await getStorefrontCountryCode();
  if (!code) {
    return false;
  }
  return code === 'USA' || code === 'US';
}
